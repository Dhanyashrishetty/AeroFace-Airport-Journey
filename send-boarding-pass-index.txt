const corsHeaders = {
  "Access-Control-Allow-Origin": "*",
  "Access-Control-Allow-Methods": "GET, POST, PUT, DELETE, OPTIONS",
  "Access-Control-Allow-Headers": "Content-Type, Authorization, X-Client-Info, Apikey",
};

interface SendEmailRequest {
  emailType?: "confirmation" | "payment-receipt" | "boarding-pass";
  passengerId: string;
  recipientEmail: string;
  passengerName: string;
  ticketNo: string;
  flightNo: string;
  origin: string;
  destination: string;
  departure: string;
  arrival: string;
  seatNo: string;
  terminal: string;
  gate: string;
  price: number;
  paymentId: string;
  /** Optional PDF attachment as base64 (no data: prefix) */
  pdfBase64?: string;
  pdfFilename?: string;
}

function esc(s: string) {
  return String(s ?? "")
    .replace(/&/g, "&amp;")
    .replace(/</g, "&lt;")
    .replace(/>/g, "&gt;")
    .replace(/"/g, "&quot;");
}

function fmt(iso: string) {
  if (!iso) return "—";
  try {
    return new Date(iso).toLocaleString("en-IN", {
      day: "2-digit",
      month: "short",
      year: "numeric",
      hour: "2-digit",
      minute: "2-digit",
    });
  } catch {
    return iso;
  }
}

async function sendWithResend(
  to: string,
  subject: string,
  text: string,
  html: string,
  pdfBase64?: string,
  pdfFilename?: string,
) {
  const apiKey = Deno.env.get("RESEND_API_KEY");
  if (!apiKey) return { sent: false, provider: null as string | null, error: "RESEND_API_KEY not set" };

  const from = Deno.env.get("RESEND_FROM_EMAIL") ?? "AeroFace <onboarding@resend.dev>";

  const payload: Record<string, unknown> = { from, to: [to], subject, text, html };
  if (pdfBase64 && pdfBase64.length > 100) {
    payload.attachments = [
      {
        filename: pdfFilename || "document.pdf",
        content: pdfBase64,
      },
    ];
  }

  const res = await fetch("https://api.resend.com/emails", {
    method: "POST",
    headers: {
      Authorization: `Bearer ${apiKey}`,
      "Content-Type": "application/json",
    },
    body: JSON.stringify(payload),
  });

  const data = await res.json().catch(() => ({}));
  if (!res.ok) {
    return { sent: false, provider: "resend", error: data?.message ?? `HTTP ${res.status}` };
  }
  return { sent: true, provider: "resend", error: null, id: data?.id };
}

Deno.serve(async (req: Request) => {
  if (req.method === "OPTIONS") {
    return new Response(null, { status: 200, headers: corsHeaders });
  }

  try {
    const body: SendEmailRequest = await req.json();
    if (!body.recipientEmail || !body.recipientEmail.includes("@")) {
      return new Response(
        JSON.stringify({ success: false, delivered: false, error: "Invalid recipient email" }),
        { status: 400, headers: { ...corsHeaders, "Content-Type": "application/json" } },
      );
    }

    const { subject, text, html, emailType } = buildContent(body);
    const defaultName =
      emailType === "boarding-pass"
        ? `BoardingPass_${body.flightNo}_${body.ticketNo}.pdf`
        : `PaymentReceipt_${body.ticketNo}.pdf`;

    const delivery = await sendWithResend(
      body.recipientEmail,
      subject,
      text,
      html,
      body.pdfBase64,
      body.pdfFilename || defaultName,
    );

    try {
      const supabase = createClient(
        Deno.env.get("SUPABASE_URL") ?? "",
        Deno.env.get("SUPABASE_SERVICE_ROLE_KEY") ?? Deno.env.get("SUPABASE_ANON_KEY") ?? "",
      );
      await supabase.from("emails").insert({
        passenger_id: body.passengerId || null,
        recipient: body.recipientEmail,
        recipient_email: body.recipientEmail,
        subject,
        body: text,
        email_type: emailType,
        pdf_attached: !!body.pdfBase64,
        status: delivery.sent ? "SENT" : "FAILED",
      });
    } catch (e) {
      console.error("emails table log failed", e);
    }

    return new Response(
      JSON.stringify({
        success: true,
        delivered: delivery.sent,
        provider: delivery.provider,
        error: delivery.error,
        message: delivery.sent
          ? `Email delivered to ${body.recipientEmail}`
          : `Email not delivered (${delivery.error}). Set RESEND_API_KEY on the edge function.`,
        subject,
        body: text,
        recipientEmail: body.recipientEmail,
        emailType,
      }),
      { status: 200, headers: { ...corsHeaders, "Content-Type": "application/json" } },
    );
  } catch (err) {
    const message = err instanceof Error ? err.message : "Unknown error";
    return new Response(
      JSON.stringify({ success: false, delivered: false, error: message }),
      { status: 500, headers: { ...corsHeaders, "Content-Type": "application/json" } },
    );
  }
});
